===== FILE: D:\prince college hackathon\client\src\context\AuthContext.jsx =====
import { createContext, useEffect, useState } from "react";
import { toast } from "react-hot-toast";
import { storage } from "../utils/storage";
import { getMe, loginUser } from "../api/auth.api";

export const AuthContext = createContext(null);

export default function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true); // checks auth on page refresh

  const loadUser = async () => {
    try {
      const token = storage.getToken();
      if (!token) {
        setUser(null);
        setLoading(false);
        return;
      }

      const data = await getMe();
      setUser(data.user);
    } catch (err) {
      storage.removeToken();
      setUser(null);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    loadUser();
  }, []);

  const login = async ({ email, password }) => {
    const data = await loginUser({ email, password });

    storage.setToken(data.token);
    setUser(data.user);

    toast.success("Logged in âœ…");
    return data;
  };

  const logout = () => {
    storage.removeToken();
    setUser(null);
    toast.success("Logged out âœ…");
  };

  const value = {
    user,
    loading,
    login,
    logout,
    loadUser,
    isLoggedIn: !!user,
    isAdmin: user?.role === "admin",
  };

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

const authMiddleware = (req, res, next) => {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader || !authHeader.startsWith("Bearer ")) {
      return res.status(401).json({ message: "No token provided" });

/**
 * Calculate distance between two geo points (meters)
 */
const getDistanceMeters = (lat1, lon1, lat2, lon2) => {
  const R = 6371000; // Earth radius in meters
  const toRad = (v) => (v * Math.PI) / 180;

  const dLat = toRad(lat2 - lat1);
  const dLon = toRad(lon2 - lon1);

  const a =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(toRad(lat1)) *
      Math.cos(toRad(lat2)) *
      Math.sin(dLon / 2) ** 2;

  return 2 * R * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
};

/**
 * Find possible duplicate issues
 */
const findDuplicateIssues = async ({
  category,
  latitude,
  longitude,
  radiusMeters = 100,
  recentDays = 7,
}) => {
  const since = new Date();
  since.setDate(since.getDate() - recentDays);

  const candidates = await Issue.find({
    category,
    status: { $ne: "closed" },
    createdAt: { $gte: since },
  });

  return candidates.filter((issue) => {
    const distance = getDistanceMeters(
      latitude,
      longitude,
      issue.latitude,
      issue.longitude
    );
    return distance <= radiusMeters;
  });
};

===== FILE: D:\prince college hackathon\server\src\modules\issue\issue.staff.controller.js =====
const Issue = require("../../models/Issue.model");

/**
 * Department staff resolves an issue
 * Proof image is mandatory
 */
const resolveIssueByStaff = async (req, res) => {
  try {
    const issueId = req.params.id;
    const staffId = req.user.userId;
    const staffDeptId = req.user.department_id;

    const { fake, fake_reason } = req.body;

    const issue = await Issue.findById(issueId);

    if (!issue) {
      return res.status(404).json({ message: "Issue not found" });
    }

    // ðŸ”’ Ensure issue belongs to staff's department
    if (
      !issue.assigned_department_id ||
      issue.assigned_department_id.toString() !== staffDeptId
    ) {
      return res.status(403).json({
        message: "You are not authorized to resolve this issue",
      });
    }

    // ðŸ”’ Ensure valid status
    if (
      issue.status !== "assigned_auto" &&
      issue.status !== "assigned"
    ) {
      return res.status(400).json({
        message: "Issue is not in a resolvable state",
      });
    }

    // ðŸ”’ Proof image is mandatory
    if (!req.file?.path) {
      return res.status(400).json({
        message: "Resolution proof image is required",
      });
    }

    issue.resolution_proof_image = req.file.path;
    issue.resolved_by = staffId;
    issue.resolved_at = new Date();
    issue.status = "resolved_pending_verification";

    // Optional fake flag by staff
    if (fake === "true") {
      issue.staff_flagged_fake = true;
      issue.staff_fake_reason = fake_reason || "No reason provided";
    }

    await issue.save();

    return res.json({
      message: "Issue marked as resolved, pending admin verification",
      issue,
    });
  } catch (error) {
    console.error("STAFF RESOLVE ISSUE ERROR:", error);
    return res.status(500).json({ message: "Server error" });
  }
};

module.exports = { resolveIssueByStaff };

  return {

      address.neighbourhood ||
      address.village ||
      address.town ||
      "",
    city: address.city || address.county || "",

  };
